<script setup>
import { ref, computed, watch } from 'vue'

const props = defineProps({
  modelValue: {
    type: String,
    default: 'RZA2030',
  },
  plateType: {
    type: String,
    default: 'private', // 'private', 'public_transport', 'private_transport', 'diplomatic', 'export'
  },
  plateSize: {
    type: String,
    default: 'long', // 'long', 'standard', 'us_square', 'table_grid'
  },
  showScrews: {
    type: Boolean,
    default: true,
  }
})

const emit = defineEmits(['update:modelValue'])

// The 17 Official Saudi Traffic Directorate Letters & Latin Transliterations
const officialLetters = [
  { ar: 'أ', lat: 'A', name: 'ألف' },
  { ar: 'ب', lat: 'B', name: 'باء' },
  { ar: 'ح', lat: 'J', name: 'حاء' },
  { ar: 'د', lat: 'D', name: 'دال' },
  { ar: 'ر', lat: 'R', name: 'راء' },
  { ar: 'س', lat: 'S', name: 'سين' },
  { ar: 'ص', lat: 'X', name: 'صاد' },
  { ar: 'ط', lat: 'T', name: 'طاء' },
  { ar: 'ع', lat: 'E', name: 'عين' },
  { ar: 'ق', lat: 'G', name: 'قاف' },
  { ar: 'ك', lat: 'K', name: 'كاف' },
  { ar: 'ل', lat: 'L', name: 'لام' },
  { ar: 'م', lat: 'Z', name: 'ميم' },
  { ar: 'ن', lat: 'N', name: 'نون' },
  { ar: 'هـ', lat: 'H', name: 'هاء' },
  { ar: 'و', lat: 'U', name: 'واو' },
  { ar: 'ى', lat: 'V', name: 'ألف مقصورة' },
]

const arabicIndicDigits = {
  '0': '٠', '1': '١', '2': '٢', '3': '٣', '4': '٤',
  '5': '٥', '6': '٦', '7': '٧', '8': '٨', '9': '٩'
}

// Letter & Digit state
const selectedLetters = ref(['R', 'Z', 'A'])
const selectedDigits = ref(['2', '0', '3', '0'])

// Parse modelValue
const parseModelValue = (val) => {
  if (!val) return
  const clean = val.toUpperCase().replace(/\s+/g, '')
  const l = clean.replace(/[^A-Z]/g, '').slice(0, 3).padEnd(3, 'A').split('')
  const d = clean.replace(/[^0-9]/g, '').slice(0, 4).padEnd(4, '0').split('')
  selectedLetters.value = l
  selectedDigits.value = d
}

watch(() => props.modelValue, (newVal) => {
  parseModelValue(newVal)
}, { immediate: true })

const updatePlate = () => {
  const code = (selectedLetters.value.join('') + selectedDigits.value.join('')).toUpperCase()
  emit('update:modelValue', code)
}

// Map Latin letters to Arabic letters
const lettersAr = computed(() => {
  return selectedLetters.value.map(lat => {
    const found = officialLetters.find(item => item.lat === lat)
    return found ? found.ar : 'أ'
  })
})

// Digits in Eastern Arabic Indic
const digitsIndic = computed(() => {
  return selectedDigits.value.map(d => arabicIndicDigits[d] || d)
})
</script>

<template>
  <div class="ksa-plate-container">
    <!-- Embossed 3D License Plate -->
    <div class="ksa-plate-metal" :class="plateSize">
      <!-- Metallic Reflection -->
      <div class="plate-reflection"></div>

      <!-- Screws -->
      <div v-if="showScrews" class="screw screw-tl"></div>
      <div v-if="showScrews" class="screw screw-tr"></div>
      <div v-if="showScrews" class="screw screw-bl"></div>
      <div v-if="showScrews" class="screw screw-br"></div>

      <!-- Main Plate Body -->
      <div class="plate-inner">
        <!-- Left: Digits -->
        <div class="plate-section plate-digits">
          <div class="digits-indic">
            <span v-for="(digit, i) in digitsIndic" :key="'ind-' + i">{{ digit }}</span>
          </div>
          <div class="digits-latin">
            <span v-for="(digit, i) in selectedDigits" :key="'lat-' + i">{{ digit }}</span>
          </div>
        </div>

        <div class="plate-divider"></div>

        <!-- Middle: Letters -->
        <div class="plate-section plate-letters">
          <div class="letters-arabic">
            <span v-for="(char, i) in lettersAr" :key="'ar-' + i">{{ char }}</span>
          </div>
          <div class="letters-latin">
            <span v-for="(char, i) in selectedLetters" :key="'latl-' + i">{{ char }}</span>
          </div>
        </div>

        <div class="plate-divider"></div>

        <!-- Right: Emblem & Kingdom Zone -->
        <div class="plate-section plate-emblem" :class="plateType">
          <div class="emblem-svg">
            <!-- Palm and Swords Emblem -->
            <svg viewBox="0 0 100 100" width="28" height="28" fill="currentColor">
              <path d="M50 8 C46 16 38 22 25 24 C34 26 42 27 48 34 C43 28 35 27 22 29 C32 33 40 37 45 44 C41 40 33 40 22 43 C33 47 43 49 47 54 C46 51 45 46 45 42 L55 42 C55 46 54 51 53 54 C57 49 67 47 78 43 C67 40 59 40 55 44 C60 37 68 33 78 29 C65 27 57 28 52 34 C58 27 66 26 75 24 C62 22 54 16 50 8 Z" />
              <path d="M47 42 L53 42 L52 58 C52 61 53 64 54 66 L46 66 C47 64 48 61 48 58 Z" />
              <path d="M18 78 C25 76 34 76 43 78 L53 81 C63 84 74 89 82 93 C78 90 70 82 60 76 C52 71 42 68 32 68 C27 68 22 69 18 71 Z" />
              <path d="M82 78 C75 76 66 76 57 78 L47 81 C37 84 26 89 18 93 C22 90 30 82 40 76 C48 71 58 68 68 68 C73 68 78 69 82 71 Z" />
            </svg>
          </div>
          <div class="emblem-code">K S A</div>
          <div class="emblem-country">السـعوديـة</div>
        </div>
      </div>
    </div>

    <!-- Enhanced Control Selectors -->
    <div class="plate-controls">
      <div class="control-group">
        <label>الأحرف (الأبجدية النظامية)</label>
        <div class="select-row">
          <select v-for="(char, idx) in selectedLetters" :key="'sel-l-' + idx"
                  v-model="selectedLetters[idx]" @change="updatePlate" class="ksa-select">
            <option v-for="item in officialLetters" :key="item.lat" :value="item.lat">
              {{ item.ar }} ({{ item.lat }})
            </option>
          </select>
        </div>
      </div>

      <div class="control-group">
        <label>الأرقام</label>
        <div class="select-row">
          <select v-for="(digit, idx) in selectedDigits" :key="'sel-d-' + idx"
                  v-model="selectedDigits[idx]" @change="updatePlate" class="ksa-select">
            <option v-for="n in 10" :key="n-1" :value="String(n-1)">
              {{ n-1 }} ({{ arabicIndicDigits[String(n-1)] }})
            </option>
          </select>
        </div>
      </div>
    </div>
  </div>
</template>
